The Dilemma of Modern Systems Software

For four decades, systems developers have been forced to choose between accidental complexity, runtime latency, or steep cognitive overhead.

01. THE C++ TAX

40 Years of Baggage

Five competing initialization styles (=, (), {}, auto = {}), accidental deep copies by default, silent dangling references from temporary strings, 400-line template error spews, and 1,000-line brittle CMake scripts just to link a library.

02. THE FALSE COMPROMISES

Rust & Go Trade-Offs

Go offers fast onboarding, but its mandatory ~20MB garbage collector and GC latency spikes disqualify it for real-time engines, audio, and kernels.

Rust delivers safety, but imposes extreme cognitive tax: fighting the borrow checker, complex lifetime annotations ('a), slow builds, and painful C++ interop walls.

03. THE CFIRE SYNTHESIS

The Missing Systems Balance

Write clean, concise code with Go-like readability, Rust-grade intent qualifiers (in, mut, out, move), and zero-cost 1-to-1 lowering to human-readable ISO C++23. Pure RAII, zero GC pauses, direct Clang header ingestion, and 1-click CMake ejection.

How cfire Radically Simplifies C++

Compare painful, error-prone C++ patterns against clean, self-documenting cfire syntax.

01. INTENT & MUTABILITY SAFETY

Call-Site Mutation Visibility

In raw C++, process(data); hides whether data is read-only or secretly modified in-place. In cfire, callers must write process(mut data), and the compiler rejects hidden side effects at compile time (cfire/MUT-CALLSITE). 

// cfire: Intent is explicit at definition and call site
fn increment(mut b: Box, amount: i32) {
    b.value = b.value + amount
}

mut my_box := Box{ value: 40 }
increment(mut my_box, 2) // OK: 'mut' marker mandatory
02. ERROR RECOVERY & DEFER

Fearless Error Propagation

C++ exceptions introduce hidden control flow and binary bloat, while std::expected requires verbose monadic chains. cfire fallible functions return T! with clean try, inline fallback or val, and deterministic defer. 

// cfire: Fallible operations with deterministic defer
fn load_config(path: str) -> Config! {
    mut f := try fs.open(path, "r")
    defer f.close() // Always runs on return or error
    payload := try f.read_all()
    return try json.parse[Config](payload)
}
03. ALGEBRAIC ENUMS & MATCH

Exhaustive Pattern Matching

In C++23, std::variant requires the cumbersome 6-line template<class... Ts> struct overloaded visitor pattern. cfire features first-class algebraic enums with concise slice and value pattern destructuring. 

// cfire: Tagged unions with exhaustive compiler checks
type Shape enum { Circle(f64), Rect(f64, f64) }

fn area(s: Shape) -> f64 {
    match s {
        Circle(r)    => { return 3.14159 * r * r }
        Rect(w, h)   => { return w * h }
    }
}
04. ZERO-FFI C/C++ INTEROP

Direct Clang AST Ingestion

Forget hand-written extern "C" wrappers, Cgo pointer marshaling, or bindgen scripts. cfire directly dumps Clang's JSON AST at compile time, caches header signatures, and automatically coerces string boundaries. 

// cfire: Ingest C/C++ headers directly with zero wrappers
import cpp "fast_hash.h"

fn main() {
    msg := "cfire systems language"
    // 'str' passes automatically as 'const char*'
    hash := cpp::fh_fnv1a(msg, msg.len())
    println("Hash: 0x{:x}", hash)
}

type Token enum {

fn classify(s: []str) -> str {
    match s {
        []                   => { return "empty" }
        [head, ..tail]       => { return "head=" + head }
        [first, ..mid, last] => { return first + ".." + last }

#include "cfire_prelude.hpp"

struct Token {
    struct _Ident {
        std::string _0{};
    };
    struct _Number {
        std::int64_t _0{};
    };
    struct _Eof {
    };
    std::variant<_Ident, _Number, _Eof> _v;
    [[nodiscard]] static auto Ident(std::string a0) -> Token { return Token{_Ident{std::move(a0)}}; }
    [[nodiscard]] static auto Number(std::int64_t a0) -> Token { return Token{_Number{std::move(a0)}}; }
    [[nodiscard]] static auto Eof() -> Token { return Token{_Eof{}}; }
};

[[nodiscard]] auto classify(const std::vector<std::string>& s) -> std::string {
    {
        const auto& _target = s;
        bool _matched = false;
        if (!_matched && _target.size() == 0) {
            _matched = true;
            return "empty";
        }
        if (!_matched && _target.size() >= 1) {
            _matched = true;
            const auto& head = _target[0];
            auto tail = cfire::slice(_target, 1, _target.size());
            return ("head=" + head);
        }
        if (!_matched && _target.size() >= 2) {
            _matched = true;
            const auto& first = _target[0];
            auto mid = cfire::slice(_target, 1, (_target.size() - 1));
            const auto& last = _target[_target.size() - 1];
            return ((first + "..") + last);
        }
    }
    return "other";
}

auto main(int argc, char** argv) -> int {
    std::vector<std::string> items = {"alpha", "beta", "gamma"};
    std::println("result: {}", classify(items));
    return 0;
}

// @cfire-link: -lsimdjson
#define CFIRE_USE_SIMDJSON 1
#include "cfire_prelude.hpp"
#include "cfire_core_stdlib.hpp"

struct Metric {
    std::string service{};
    double latency{};
    bool healthy{};
};

template <>
struct cfire::reflect_struct<Metric> {
    static constexpr std::string_view name = "Metric";
    static constexpr std::size_t field_count = 3;
    static constexpr std::array<std::string_view, 3> field_names = {
        "service",
        "latency",
        "healthy",
    };
    static constexpr std::array<std::string_view, 3> field_tags = {
        R"tag(json:"service")tag",
        R"tag(json:"latency_ms")tag",
        R"tag(json:"healthy")tag",
    };
};

template <>
inline auto cfire::json::parse_struct<Metric>(simdjson::ondemand::object obj) -> std::expected<Metric, cfire::Error> {
    Metric out{};
    CFIRE_TRY_ASSIGN(out.service, cfire::json::get_str(obj, "service"));
    CFIRE_TRY_ASSIGN(out.latency, cfire::json::get_f64(obj, "latency_ms"));
    CFIRE_TRY_ASSIGN(out.healthy, cfire::json::get_bool(obj, "healthy"));
    return out;
}

template <>
inline auto cfire::json::stringify_struct<Metric>(const Metric& val) -> std::string {
    std::string out = "{";
    out += std::format("\"service\":\"{}\"", cfire::json::escape_string(val.service));
    out += std::format(",\"latency_ms\":{}", val.latency);
    out += std::format(",\"healthy\":{}", val.healthy ? "true" : "false");
    out += "}";
    return out;
}

auto main(int argc, char** argv) -> int {
    auto payload = R"({"service":"auth","latency_ms":1.42,"healthy":true})";
    auto _try_1 = cfire::json::parse<Metric>(payload);
    if (!_try_1.has_value()) {
        std::println(stderr, "Error: {}", _try_1.error().message());
        return 1;
    }
    auto m = std::move(*_try_1);
    std::println("service={} latency={}ms", m.service, m.latency);
    return 0;
}

Get Started in 3 Steps

From zero to running your first high-performance systems service in under 60 seconds.

STEP 1

Build or Install

Clone and compile the pure C++23 compiler in seconds with zero external dependencies:

git clone https://github.com/gsandoval/cfire
cd cfire && make
STEP 2

Write `main.cf`

Create a clean, idiomatic service file with built-in stdlib:

package main

import "core:time"

fn main() {
    now := time.now()
    println("Hello from cfire! 🚀")
    println("Time: {}", now.format_rfc3339())
}
STEP 3

Run, Build or Eject

Execute immediately, compile a binary, or eject to CMake:

./cfire run main.cf
./cfire build main.cf -o my_service
./cfire eject main.cf ./standalone_pkg
